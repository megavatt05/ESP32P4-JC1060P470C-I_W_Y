# ============================================================================
#  Сценарий сборки главного компонента (main) проекта jd9165_hello_display
# ----------------------------------------------------------------------------
#  Файлы проекта:
#    main.c            — инициализация дисплея JD9165, сплэш, цикл анимации;
#    ui.c / ui.h       — примитивы UI: заливка, рамки, линии, UTF-8 текст,
#                        панели, кнопки, полосы прогресса;
#    ui_demo.c/.h      — демо-экран 1024x600 и сплэш;
#    ui_font_sans16.c  — готовый шрифт DejaVu Sans 16 px (кириллица);
#    ui_font_sans24b.c — готовый шрифт DejaVu Sans Bold 24 px;
#    ui_font_sans40b.c — готовый шрифт DejaVu Sans Bold 40 px
#    (шрифты генерируются скриптом tools/make_font.py, уже включены в репо).
#  Драйвер панели, esp_lcd и драйверы LDO/LEDC подтягиваются из ESP-IDF
#  и реестра компонентов (см. idf_component.yml).
# ============================================================================

# Регистрация компонента: все исходники, текущая папка — в путях поиска
idf_component_register(
    SRCS "main.c"
         "ui.c"
         "ui_demo.c"
         "ui_font_sans16.c"
         "ui_font_sans24b.c"
         "ui_font_sans40b.c"
    INCLUDE_DIRS "."
)
